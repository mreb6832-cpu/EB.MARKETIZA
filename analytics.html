import { supabase } from "./supabaseClient.js"; const SESSION_KEY = "eb_analytics_session"; let sessionId = sessionStorage.getItem(SESSION_KEY); if (!sessionId) { sessionId = crypto.randomUUID(); sessionStorage.setItem(SESSION_KEY, sessionId); } async function trackVisitor() { const { error } = await supabase .from("visitor_events") .insert({ session_id: sessionId, page_path: window.location.pathname }); if (error) { console.error("Visitor tracking error:", error.message); } } async function updateVisitorStats() { const { data, error } = await supabase.rpc( "get_visitor_stats" ); if (error) { console.error("Visitor stats error:", error.message); return; } const stats = data || {}; const online = document.getElementById("visitor-online"); const today = document.getElementById("visitor-today"); const total = document.getElementById("visitor-total"); if (online) online.textContent = stats.online ?? 0; if (today) today.textContent = stats.today ?? 0; if (total) total.textContent = stats.total ?? 0; } // Record initial visit trackVisitor(); updateVisitorStats(); // Keep online visitors active setInterval(trackVisitor, 4 * 60 * 1000); // Refresh displayed numbers setInterval(updateVisitorStats, 30 * 1000);
